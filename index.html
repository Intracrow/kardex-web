function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Hoja 1") || ss.getSheets()[0]; const data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date(), data.nombre || "", data.telefono || "", data.correo || "", data.tipoMueble || "", data.modeloReferencia || "", data.material || "", data.aclaraciones || "" ]); return ContentService .createTextOutput(JSON.stringify({ status: "ok" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: err.message })) .setMimeType(ContentService.MimeType.JSON); } }
